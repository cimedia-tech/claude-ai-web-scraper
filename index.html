GitHub Actions Scheduler

.github/workflows/scraper.yml
name: Claude Scraper Job
on:
  schedule:
    - cron: '0 6 * * *'       # 06:00 UTC daily
  workflow_dispatch:          # Manual trigger button
jobs:
  scrape:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-python@v5
        with:
          python-version: '3.12'
      - run: pip install -r requirements.txt && crawl4ai-setup
      - run: python scraper.py
        env:
          ANTHROPIC_API_KEY: ${{ secrets.ANTHROPIC_API_KEY }}
      - uses: actions/upload-artifact@v4
        with:
          name: scraped-data
          path: output/*.json

10Reference Implementation

Core Code Patterns

Pattern A — Async Extraction (Crawl4AI + Claude Haiku)

scraper.py · Python 3.12
import asyncio, json
from crawl4ai import AsyncWebCrawler, BrowserConfig, CrawlerRunConfig
from anthropic import Anthropic
from pydantic import BaseModel

client = Anthropic()

class LeadData(BaseModel):
    company: str
    contact: str | None
    email: str | None

async def extract(url: str) -> list[LeadData]:
    cfg = CrawlerRunConfig(
        word_count_threshold=10,
        excluded_tags=["script", "style", "nav", "footer"],
        remove_overlay_elements=True,
    )
    async with AsyncWebCrawler(config=BrowserConfig(headless=True)) as c:
        r = await c.arun(url=url, config=cfg)
        content = r.markdown[:8000]  # Token gate

    resp = client.messages.create(
        model="claude-haiku-4-5",
        max_tokens=1024,
        system={"type":"text","text":"Extract leads as JSON array.",
                "cache_control":{"type":"ephemeral"}},
        messages=[{"role":"user","content":content}]
    )
    raw = json.loads(resp.content[0].text)
    return [LeadData(**i) for i in raw]

asyncio.run(extract("https://example.com/directory"))
